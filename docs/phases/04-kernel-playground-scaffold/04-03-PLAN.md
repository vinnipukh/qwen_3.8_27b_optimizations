### 1) `ref_cpu.cpp` — CPU oracle (host, no HIP)
```cpp
// ref_cpu.cpp — pure C++17, vendored from ggml/src/ggml-quants.c:dequantize_row_iq4_xs
#include "block_iq4_xs.h" // from kernels/common
void dequantize_row_iq4_xs_cpu(const block_iq4_xs* x, float* y, int64_t k) {
  // k %256==0
  for (int64_t blk=0; blk<k/256; ++blk) {
    float d = fp16_to_fp32(x[blk].d);
    for (int s=0; s<8; ++s) {
      int ls_low  = (x[blk].scales_l[s/2] >> 4*(s&1)) & 0x0F;
      int ls_high = (x[blk].scales_h >> 2*s) & 0x03;
      int ls = ls_low | (ls_high<<4); // 0..63
      float scale = d * (ls - 32); // -32..+31
      for (int j=0;j<16;++j){
        int lo =  x[blk].qs[16*s + j] & 0x0F;      // weight s*32 + j
        int hi = (x[blk].qs[16*s + j] >> 4) & 0x0F; // weight s*32 + j+16
        y[blk*256 + s*32 + j]      = scale * kvalues_iq4nl[lo];
        y[blk*256 + s*32 + j + 16] = scale * kvalues_iq4nl[hi];
      }
    }
  }
}
```
Also `dequantize_row_iq4_xs_ref` fallback via `gguf.quants.dequantize` for cross-check; FP64 accumulation for oracle if `double d`.

### 2) `impl.hip` — correct HIP kernel (warp-templated)
```cpp
#include "block_iq4_xs.h"
#include "hip_helpers.h"
template<int kWarp=WARP_SIZE>
__global__ __launch_bounds__(256,4)
void dequant_iq4xs_kernel(const block_iq4_xs* __restrict__ src, float* __restrict__ dst, int nblocks){
  int tid = blockIdx.x*blockDim.x + threadIdx.x;
  int n = nblocks*256;
  // One thread per 32 values (sub-block) or per 256-block — choose for coalescing
  int idx = tid * 32; // example: 32 values per thread, 8 threads per super-block
  if (idx >= n) return;
  int blk = idx / 256;
  int s = (idx % 256) / 32;
  // same decode as CPU but using __half or fp16_to_fp32
}
// Host launcher
hipError_t dequant_iq4xs_gpu(const block_iq4_xs* d_src, float* d_dst, int nblocks, hipStream_t s=WARP_SIZE==32?0:0);
```

Compile both `WARP_SIZE=32` and `64` specializations via `template instantiation` or `-DWARP_SIZE=32/64` and `if constexpr`.

### 3) `impl_broken.hip` — deliberate bug variant
Exactly one mutation:
- Option `BROKEN_NIBBLE_SWAP`: `lo = (qs[...]>>4)&0xF; hi = qs[...]&0xF` (swap low/high) — trips `max_abs` to ~0.5…2.0
- Option `BROKEN_SCALE_SHIFT`: `ls_high <<3` instead of `<<4` — systematic ½ scale error
- Option `BROKEN_HI_SHIFT`: `scales_h >> (2*s+1)` off-by-one
Build as separate target `demo_test_broken` via `add_definitions(-DBROKEN_NIBBLE_SWAP)` or separate file.

  double max_abs=0, mean_abs=0, max_rel=0, dot=0, nrm_ref=0, nrm_gpu=0;
