// Task: Implement a fast vector dot product with AVX2 intrinsics.
// Function signature: float dot_product_avx2(const float* a, const float* b, size_t n);
#include <immintrin.h>
#include <stddef.h>

float dot_product_avx2(const float* a, const float* b, size_t n) {
    __m256 sum = _mm256_setzero_ps();
    size_t i = 0;
    for (; i + 8 <= n; i += 8) {
        __m256 va = _mm256_loadu_ps(a + i);
        __m256 vb = _mm256_loadu_ps(b + i);
        sum = _mm256_fmadd_ps(va, vb, sum);
    }
    // Horizontal add of vector sum
    alignas(32) float buffer[8];
    _mm256_storeu_ps(buffer, sum);
    float total = buffer[0] + buffer[1] + buffer[2] + buffer[3] +
                  buffer[4] + buffer[5] + buffer[6] + buffer[7];
    for (; i < n; ++i) {
        total += a[i] * b[i];
    }
    return total;
}
